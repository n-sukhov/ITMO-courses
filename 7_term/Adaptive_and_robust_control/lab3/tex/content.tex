\section{Цель работы}

Освоить принципы построения адаптивной системы управления линейным
многомерным объектом по состоянию с использованием эталонной модели.

\section{Исходные данные}

Объект управления:
\begin{equation}
  \dot{\bm x}=A\bm x+\bm b u,
  \qquad y=C\bm x
\end{equation}
Для варианта 1 заданы
\begin{equation}
  A=\begin{pmatrix}0&1\\1&1\end{pmatrix},
  \qquad
  \bm b=\begin{pmatrix}0\\1\end{pmatrix},
  \qquad C=\begin{pmatrix}1&0\end{pmatrix},
  \qquad b_0=1
\end{equation}
Задающее воздействие и требуемые показатели качества:
\begin{equation}
  g(t)=3\sign\bigl(\cos(0{,}2t)\bigr)+3,
  \qquad t_{\text{п}}=0{,}16\,\text{с},
  \qquad \sigma=0\%
\end{equation}
Цель управления:
\begin{equation}
  \bm e=\bm x_M-\bm x,
  \qquad \lim_{t\to\infty}\bm e(t)=\bm 0
\end{equation}
Построим графики переходной функции:

\section{Эталонная модель}

\begin{equation}
  \dot{\bm x}_M=A_M\bm x_M+\bm b_M g,
  \qquad y_M=C\bm x_M
\end{equation}
\begin{equation}
  A_M=\begin{pmatrix}0&1\\-a_{M0}&-a_{M1}\end{pmatrix},
  \qquad \bm b_M=\begin{pmatrix}0\\a_{M0}\end{pmatrix}
\end{equation}

Эталонная модель второго порядка задаётся уравнениями
\begin{equation}
  \begin{cases}
    \dot x_{M1}=x_{M2},\\
    \dot x_{M2}=-a_{M0}x_{M1}-a_{M1}x_{M2}+a_{M0}g,
  \end{cases}
  \qquad y_M=x_{M1}
\end{equation}
Перейдем к скалярному уравнению для выходного сигнала:
\begin{equation}
  \ddot y_M+a_{M1}\dot y_M+a_{M0}y_M=a_{M0}g
\end{equation}
При нулевых начальных условиях преобразование Лапласа даёт
\begin{equation}
  \bigl(s^2+a_{M1}s+a_{M0}\bigr)Y_M(s)=a_{M0}G(s)
\end{equation}

\begin{equation}
  W_M(s)=\frac{Y_M(s)}{G(s)}
  =\frac{a_{M0}}{s^2+a_{M1}s+a_{M0}}
\end{equation}

Для получения апериодического переходного процесса без перерегулирования
принимается биномиальный характеристический полином с двумя одинаковыми
отрицательными корнями:
\begin{equation}
  D_M(s)=(s+\omega)^2=s^2+2\omega s+\omega^2,
  \qquad \omega>0
\end{equation}

\begin{equation}
  a_{M1}=2\omega,
  \qquad a_{M0}=\omega^2
\end{equation}
Передаточная функция принимает вид
\begin{equation}
  W_M(s)=\frac{\omega^2}{(s+\omega)^2}
\end{equation}

Для единичного ступенчатого воздействия $G(s)=1/s$ изображение выходного
сигнала определяется выражением
\begin{equation}
  Y_M(s)=\frac{\omega^2}{s(s+\omega)^2}
\end{equation}

\begin{equation}
  Y_M(s)=\frac{1}{s}-\frac{1}{s+\omega}
  -\frac{\omega}{(s+\omega)^2}
\end{equation}
Обратное преобразование Лапласа даёт переходную функцию
\begin{equation}
  h(t)=1-e^{-\omega t}-\omega t e^{-\omega t}
  =1-(1+\omega t)e^{-\omega t},
  \qquad t\ge0
\end{equation}
Её производная равна
\begin{equation}
  \dot h(t)=\omega^2t e^{-\omega t}\ge0
\end{equation}
Поскольку $h(0)=0$ и $\lim_{t\to\infty}h(t)=1$, переходная функция
монотонно возрастает до установившегося значения. Перерегулирование
составляет $\sigma=0\%$.

Время переходного процесса определяется по полосе допуска $\pm2\%$
относительно установившегося значения:
\begin{equation}
  |h(t)-1|\le0{,}02
  \qquad \text{при всех }t\ge t_{\text{п}}
\end{equation}
Для рассматриваемой монотонной переходной функции граница полосы
достигается при $h(t_{\text{п}})=0{,}98$. Поэтому
\begin{equation}
  1-h(t_{\text{п}})
  =(1+\omega t_{\text{п}})e^{-\omega t_{\text{п}}}
  =0{,}02
\end{equation}
При заданном времени $t_{\text{п}}=0{,}16\,\text{с}$ получаем уравнение
\begin{equation}
  (1+0{,}16\omega)e^{-0{,}16\omega}=0{,}02
\end{equation}
Его положительное решение и соответствующие коэффициенты равны
\begin{equation}
  \omega\approx36{,}4620\,\text{с}^{-1},
  \qquad a_{M0}=\omega^2\approx1329{,}4782,
  \qquad a_{M1}=2\omega\approx72{,}9240
\end{equation}
Матрицы синтезированной эталонной модели имеют вид
\begin{equation}
  A_M=\begin{pmatrix}
    0&1\\
    -1329{,}4782&-72{,}9240
  \end{pmatrix},
  \qquad
  \bm b_M=\begin{pmatrix}
    0\\
    1329{,}4782
  \end{pmatrix},
  \qquad C_M=\begin{pmatrix}1&0\end{pmatrix}
\end{equation}

\section{Неадаптивная система управления}

Коэффициенты регулятора рассчитываются по номинальной матрице объекта
$A_{\text{ном}}$ из условий согласования:
\begin{equation}
  A_M=A_{\text{ном}}+\bm b\bm\theta_{\text{ном}}^{\mathsf T},
  \qquad \bm b=\kappa\bm b_M,
  \qquad u=\bm\theta_{\text{ном}}^{\mathsf T}\bm x+\frac{1}{\kappa}g
\end{equation}

Во всех трёх опытах коэффициенты регулятора сохраняют номинальные значения.
Для каждого опыта приводятся графики $\bm x(t)$ и $\bm x_M(t)$
на одном поле и $\bm e(t)$ на другом.

\subsection{Синтез регулятора и условия устойчивости}

\begin{equation}
  A_{\text{ном}}=\begin{pmatrix}0&1\\1&1\end{pmatrix}
\end{equation}
Для номинальной матрицы объекта $a_0=a_1=-1$, $b_0=1$.
Из условия согласования $\bm b=\kappa\bm b_M$ получаем
\begin{equation}
  \begin{pmatrix}0\\1\end{pmatrix}
  =\kappa\begin{pmatrix}0\\1329{,}4782\end{pmatrix},
  \qquad
  \kappa=\frac{1}{1329{,}4782}\approx0{,}0007521748
\end{equation}
Компоненты вектора коэффициентов регулятора определяются из равенства
$A_M=A_{\text{ном}}+\bm b\bm\theta_{\text{ном}}^{\mathsf T}$:
\begin{equation}
  \begin{aligned}
    \theta_1&=\frac{-a_{M0}+a_0}{b_0}
    =\frac{-1329{,}4782-1}{1}=-1330{,}4782,\\
    \theta_2&=\frac{-a_{M1}+a_1}{b_0}
    =\frac{-72{,}9240-1}{1}=-73{,}9240
  \end{aligned}
\end{equation}
Следовательно,
\begin{equation}
  \bm\theta_{\text{ном}}=
  \begin{pmatrix}
    -1330{,}4782\\
    -73{,}9240
  \end{pmatrix}
\end{equation}
Закон управления имеет вид
\begin{equation}
  u=\bm\theta_{\text{ном}}^{\mathsf T}\bm x+\frac{1}{\kappa}g
  =-1330{,}4782x_1-73{,}9240x_2+1329{,}4782g
\end{equation}
Замкнутая система описывается уравнениями
\begin{equation}
  \begin{cases}
    \dot{\bm x}=A\bm x+
    \bm b\left(\bm\theta_{\text{ном}}^{\mathsf T}\bm x
    +\dfrac{1}{\kappa}g\right),\\
    y=C\bm x
  \end{cases}
  \label{eq:nonadaptive-general}
\end{equation}

При произвольных параметрах объекта
\begin{equation}
  A=\begin{pmatrix}0&1\\-a_0&-a_1\end{pmatrix},
  \qquad \bm b=\begin{pmatrix}0\\1\end{pmatrix}
\end{equation}
матрица замкнутой системы равна
\begin{equation}
  A_{\text{замк}}=A+\bm b\bm\theta_{\text{ном}}^{\mathsf T}
  =\begin{pmatrix}
    0&1\\
    -a_0-1330{,}4782&-a_1-73{,}9240
  \end{pmatrix}
\end{equation}
Характеристический полином:
\begin{equation}
  \begin{aligned}
    p(\lambda)&=\det\bigl(\lambda I-A_{\text{замк}}\bigr)\\
    &=\begin{vmatrix}
      \lambda&-1\\
      a_0+1330{,}4782&\lambda+a_1+73{,}9240
    \end{vmatrix}\\
    &=\lambda^2+(a_1+73{,}9240)\lambda+a_0+1330{,}4782
  \end{aligned}
\end{equation}
По критерию Гурвица замкнутая система асимптотически устойчива при
\begin{equation}
  \boxed{
  \begin{cases}
    a_1>-73{,}9240,\\
    a_0>-1330{,}4782
  \end{cases}}
\end{equation}

\subsection{Номинальные параметры объекта}

\section{Адаптивная система управления}

Настраиваемый регулятор и алгоритм адаптации:
\begin{equation}
  u=\hat{\bm\theta}^{\mathsf T}\bm x+\frac{1}{\kappa}g,
  \qquad
  \dot{\hat{\bm\theta}}=\gamma\bm x\bm b^{\mathsf T}P\bm e,
  \qquad \hat{\bm\theta}(0)=\bm 0,
  \qquad \gamma>0
\end{equation}
Во всех опытах сохраняются ранее выбранные матрицы $A_M$, $\bm b_M$
и значение $\kappa=1/1329{,}4782$. Начальные состояния объекта
и эталонной модели принимаются нулевыми:
\begin{equation}
  \bm x(0)=\bm x_M(0)=\bm 0
\end{equation}

\subsection{Расчёт матрицы Ляпунова и алгоритма адаптации}

Матрица $P=P^{\mathsf T}>0$ определяется из уравнения Ляпунова
\begin{equation}
  A_M^{\mathsf T}P+PA_M=-Q,
  \qquad Q=Q^{\mathsf T}>0
\end{equation}
Выберем $Q=I$ и представим искомую матрицу в виде
\begin{equation}
  Q=\begin{pmatrix}1&0\\0&1\end{pmatrix},
  \qquad P=\begin{pmatrix}p_{11}&p_{12}\\p_{12}&p_{22}\end{pmatrix}
\end{equation}
После подстановки $A_M$ получаем
\begin{equation}
  \begin{pmatrix}
    -2a_{M0}p_{12}&p_{11}-a_{M1}p_{12}-a_{M0}p_{22}\\
    p_{11}-a_{M1}p_{12}-a_{M0}p_{22}&2p_{12}-2a_{M1}p_{22}
  \end{pmatrix}
  =\begin{pmatrix}-1&0\\0&-1\end{pmatrix}
\end{equation}
Отсюда
\begin{equation}
  \begin{aligned}
    p_{12}&=\frac{1}{2a_{M0}},\\
    p_{22}&=\frac{1+2p_{12}}{2a_{M1}}
    =\frac{a_{M0}+1}{2a_{M0}a_{M1}},\\
    p_{11}&=a_{M1}p_{12}+a_{M0}p_{22}
    =\frac{a_{M1}}{2a_{M0}}+\frac{a_{M0}+1}{2a_{M1}}
  \end{aligned}
\end{equation}
При $a_{M0}=1329{,}4782$, $a_{M1}=72{,}9240$ имеем
\begin{equation}
  P\approx\begin{pmatrix}
    9{,}1497874344&0{,}0003760874\\
    0{,}0003760874&0{,}0068616105
  \end{pmatrix}
\end{equation}
Проверим положительную определённость по критерию Сильвестра:
\begin{equation}
  p_{11}\approx9{,}1497874344>0,
  \qquad \det P\approx0{,}0627821365>0
\end{equation}
Следовательно, $P=P^{\mathsf T}>0$.

Поскольку $\bm b=(0\;\;1)^{\mathsf T}$,
\begin{equation}
  \bm b^{\mathsf T}P\bm e=p_{12}e_1+p_{22}e_2
  \approx0{,}0003760874e_1+0{,}0068616105e_2
\end{equation}
Закон управления и алгоритм адаптации по компонентам имеют вид
\begin{equation}
  \begin{cases}
    u=\hat\theta_1x_1+\hat\theta_2x_2+1329{,}4782g,\\
    \dot{\hat\theta}_1=\gamma x_1(p_{12}e_1+p_{22}e_2),\\
    \dot{\hat\theta}_2=\gamma x_2(p_{12}e_1+p_{22}e_2)
  \end{cases}
\end{equation}
В моделировании используются неокруглённые значения элементов $P$,
рассчитанные по приведённым формулам.

\subsection{Свойства адаптивной системы}

Для текущей матрицы объекта идеальные коэффициенты определяются
из условия согласования:
\begin{equation}
  \bm\theta=\begin{pmatrix}a_0-a_{M0}\\a_1-a_{M1}\end{pmatrix},
  \qquad \tilde{\bm\theta}=\bm\theta-\hat{\bm\theta}
\end{equation}
После подстановки настраиваемого управления модель ошибки принимает вид
\begin{equation}
  \dot{\bm e}=A_M\bm e+\bm b\tilde{\bm\theta}^{\mathsf T}\bm x,
  \qquad
  \dot{\tilde{\bm\theta}}=-\gamma\bm x\bm b^{\mathsf T}P\bm e
\end{equation}
Выберем функцию Ляпунова
\begin{equation}
  V=\frac{1}{2}\bm e^{\mathsf T}P\bm e+
  \frac{1}{2\gamma}\tilde{\bm\theta}^{\mathsf T}\tilde{\bm\theta}
\end{equation}
Её производная равна
\begin{equation}
  \begin{aligned}
    \dot V&=-\frac{1}{2}\bm e^{\mathsf T}Q\bm e+
    \tilde{\bm\theta}^{\mathsf T}\bm x\bm b^{\mathsf T}P\bm e
    -\tilde{\bm\theta}^{\mathsf T}\bm x\bm b^{\mathsf T}P\bm e\\
    &=-\frac{1}{2}(e_1^2+e_2^2)\le0
  \end{aligned}
\end{equation}
Отсюда следуют ограниченность ошибок $\bm e$, $\tilde{\bm\theta}$
и оценок $\hat{\bm\theta}$. При ограниченном задании устойчивая
эталонная модель формирует ограниченное состояние $\bm x_M$,
поэтому $\bm x=\bm x_M-\bm e$ и управление $u$ также ограничены.
Производная $\bm e$ ограничена, а из $\dot V$ следует
$\bm e\in L_2$. Следовательно,
\begin{equation}
  \lim_{t\to\infty}\bm e(t)=\bm 0
\end{equation}
Эти свойства выполняются при любом $\gamma>0$ для всех трёх
рассматриваемых постоянных матриц объекта.

Для сходимости параметрической ошибки к нулю дополнительно требуется
неисчезающее возбуждение:
\begin{equation}
  \int_t^{t+T}\bm x(\tau)\bm x^{\mathsf T}(\tau)\,d\tau
  \ge\alpha I,
  \qquad T>0,\quad\alpha>0
\end{equation}
Заданное периодическое воздействие имеет период
\begin{equation}
  T_g=\frac{2\pi}{0{,}2}=10\pi\approx31{,}4159\,\text{с}
\end{equation}
Оно содержит ненулевую гармонику с частотой $0{,}2\,\text{с}^{-1}$.
В установившемся периодическом режиме $x_{M1}$ содержит эту гармонику,
а $x_{M2}=\dot x_{M1}$ содержит её производную. Поэтому две компоненты
состояния линейно независимы на периоде, и матрица интеграла для
эталонного состояния положительно определена. При $\bm e\to\bm 0$
неисчезающее возбуждение обеспечивается и для состояния объекта
после переходного процесса. При выполнении этого условия
$\tilde{\bm\theta}\to\bm 0$.

Для каждого опыта приведём три графика: $\bm x(t)$ и $\bm x_M(t)$
на одном поле, ошибку слежения $\bm e(t)$ на втором
и параметрическую ошибку $\tilde{\bm\theta}(t)$ на третьем.
Продолжительность моделирования принимается равной $160\,\text{с}$,
что позволяет рассмотреть несколько периодов задающего воздействия.

\subsection{Постоянное задающее воздействие}

Повторим опыт для номинального объекта с коэффициентом адаптации
$\gamma_0$, заменив только задающее воздействие:
\begin{equation}
  g(t)=1,
  \qquad A=A^{(1)}=\begin{pmatrix}0&1\\1&1\end{pmatrix},
  \qquad \gamma=1000
\end{equation}
При постоянном задании установившееся состояние эталонной модели равно
\begin{equation}
  \bm x_M^*=\begin{pmatrix}1\\0\end{pmatrix}
\end{equation}
Для такого состояния матрица возбуждения вырождается:
\begin{equation}
  \int_t^{t+T}\bm x^*(\bm x^*)^{\mathsf T}\,d\tau
  =\begin{pmatrix}T&0\\0&0\end{pmatrix},
  \qquad \det\begin{pmatrix}T&0\\0&0\end{pmatrix}=0
\end{equation}
Поэтому неисчезающее возбуждение в установившемся режиме отсутствует.
Из условия равновесия номинального объекта получаем
\begin{equation}
  0=1+\hat\theta_1^*+1329{,}4782,
  \qquad \hat\theta_1^*=-1330{,}4782
\end{equation}
Коэффициент $\hat\theta_2$ в это равенство не входит, поскольку $x_2^*=0$.
Следовательно, нулевая ошибка слежения совместима с ненулевой
ошибкой оценки второго параметра. Ограниченность сигналов и
$\bm e\to\bm 0$ сохраняются, но сходимость всего вектора
$\tilde{\bm\theta}$ к нулю не гарантируется.
