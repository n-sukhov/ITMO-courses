\section{Исходные данные}
\begin{itemize}
    \item Модель транзистора – 2N3904;
    \item $E=5$В, $R_1=R_4=1$кОм, $R_2=R_3=50$кОм, $T=2$с;
    \item При $C_1 = C_2 = C$ и $R_2 = R_3 = R$ значение емкости конденсатора по формуле: $T = 0.7(R C + R C) = 0.7 \cdot 2R C = 1.4 R C$
    $C = \frac{T}{1.4 R} = \frac{2}{1.4 \cdot 50 \cdot 10^3} = \frac{2}{70 \cdot 10^3}= 28.5714\text{мкФ}$\\
    $C_1 = C_2 =28.5714\text{мкФ}$
\end{itemize}
