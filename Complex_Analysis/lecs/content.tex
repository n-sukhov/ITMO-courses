\textbf{Теорема Морера}\\
Пусть $f$ непрерывна на откр. $D$ и $\forall \Delta\subset D \int_{\delta\Delta}{f(z)dz}=0$
Тогда $f\in\Theta(D)$.\\
\textit{Док-во}\\
Аналогично ($\exists$ первообр. у гол. ф-ии $f$ в от. области)\\

\textbf{Теорема (нули голоморфной функции)}\\
Пусть $f\in\Theta(D)$ и $f(a)=0, a\in D$.
\begin{itemize}
    \item Либо $\exists U(a): f\equiv 0$ в $U(a)$
    \item Либо $\exists U(a): f\neq 0$ в $\overset{\circ}{U}(a)$
\end{itemize}

$f(z)=
\begin{cases}
    \sin(\frac{1}{z})\cdot z, z\neq 0\\
    0, z = 0
\end{cases}
$\\

\textit{Док-во}\\
$f(z)=\sum_{k=0}^{\infty}{\frac{f^{(k)}}{k!}(z-a)^k}$ в 
$|z-a|<R$\\
1) $f^{(k)}(a)=0\Rightarrow f\equiv 0$ в $|z-a|<R$\\
иначе 2) $f^{0}(a)=0\Rightarrow \exists k_0: f^{(k_0)}(a)\neq 0$

$$
f(z)=(z-a)^{k_0}\left( \frac{f^{(k_0)}(a)}{k_0!}+\frac{f^{(k_0+1)}(a)}{(k_0+1)!}(z-a)+\dots \right)
$$
$g(z)$ непр. в $z_0$.

\textbf{Теорема (принцип аналитического продолжения)}\\
Пусть $f,g\in \Theta(D), D$ - открытое связное\\
Пусть $\exists z_k\in D: \lim_{k\to\infty}{z_k}\in D$ и $f(z_k)=g(z_k)$
и эл-ты $z_k$ различны.\\
Тогда $f\equiv g$ на $D$.\\
\textit{Док-во}\\
Рассмотрим $h(z)=f(z)-g(z)$     $\mathbf{(h(z_k)=0)}$\\
Пусть $\lim_{k\to\infty}{z_k}=a\Rightarrow h(a)=0$, причем
$h\in\Theta (D) \Rightarrow \exists r:$ в $|z-a|<r\quad h(z)=0$

\section{Ряды Лорана. Классификация изолированных особых точек}
\textbf{Теорема (Лорана)}\\
Пусть $f\in \Theta(K)$, $K=\{r<|z-z_0|<R\}, 0\leq r <R\leq +\infty$\\
Тогда $\forall z\in K \exists u!$ разложение
$f(z)=\sum_{k=-\infty}^{+\infty}c_k(z-z_0)^k$, где
$c_k=\frac{1}{2\pi i }\int_{\gamma}\frac{f(z)}{(z-z_0)^{k+1}}dz$,
$\gamma$ - окр-ть рад. от $r$ до $R$.\\
\textit{Док-во}\\
Пусть $z\in K$

$r<r'<R'<R$ и $z\in\{r'<|z-z_0|<R'\}$\\
$f(z)=\frac{1}{2\pi i }\int_{\gamma C_{R'}}\frac{f(t)}{t-z}dt-\frac{1}{2\pi i }\int_{\gamma C_{r'}}\frac{f(t)}{t-z}dt$
$\int_{\gamma C_{R'}}\frac{f(t)}{t-z}dt=\int_{\gamma C_{R'}}\frac{f(t)}{(t-z_0)+(z_0-z)}dt=
\int_{\gamma C_{R'}}\frac{1}{t-z_0}\frac{f(t)}{1-\frac{z-z_0}{t-z_0}}dt=\\
=\int_{\gamma C_{R'}}\frac{f(t)}{t-z_0}\sum_{0}^{\infty}\left( \frac{z-z_0}{t-z_0} \right)^k dt=
\sum_{0}^{\infty} (z-z_0)^k\int_{\gamma C_{R'}}\frac{f(t)}{(t-z_0^{k+1})}dt$\\
Итого: $C_k=\frac{1}{2\pi i}\int_{\gamma}\frac{f(t)}{(z-z_0)^{k+1}}dt,\quad k\geq0$
